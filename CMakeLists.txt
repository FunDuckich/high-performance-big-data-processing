cmake_minimum_required(VERSION 3.20)
project(concurrency_labs VERSION 1.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use MSVC and a Windows x64 generator.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(lab_common INTERFACE)
target_include_directories(lab_common INTERFACE "${PROJECT_SOURCE_DIR}/common")
target_compile_definitions(lab_common INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(lab_common INTERFACE /W4 /permissive- /utf-8 /EHsc /fp:precise)

foreach(lab IN ITEMS lab01 lab02 lab03)
    add_executable(${lab} "${lab}/main.cpp")
    target_link_libraries(${lab} PRIVATE lab_common)
    set_target_properties(${lab} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/${lab}/bin/$<CONFIG>")
    install(TARGETS ${lab} RUNTIME DESTINATION "${lab}/bin/Release" CONFIGURATIONS Release)
    install(FILES "${lab}/run.bat" "${lab}/README.md" DESTINATION "${lab}")
endforeach()

include(CTest)
if(BUILD_TESTING)
    add_executable(lab_tests tests/main.cpp)
    target_link_libraries(lab_tests PRIVATE lab_common)
    add_test(NAME numeric_checks COMMAND lab_tests)
    foreach(lab IN ITEMS lab01 lab02 lab03)
        add_test(NAME ${lab}_small COMMAND ${lab} --size 16 --repeats 2)
        add_test(NAME ${lab}_one COMMAND ${lab} --size 1)
        add_test(NAME ${lab}_help COMMAND ${lab} --help)
        foreach(arg IN ITEMS 0 -1 abc 999999999999999999999999)
            add_test(NAME ${lab}_invalid_${arg} COMMAND ${lab} --size ${arg})
            set_tests_properties(${lab}_invalid_${arg} PROPERTIES WILL_FAIL TRUE)
        endforeach()
        add_test(NAME ${lab}_missing_value COMMAND ${lab} --size)
        add_test(NAME ${lab}_invalid_repeats COMMAND ${lab} --repeats 0)
        add_test(NAME ${lab}_unknown_option COMMAND ${lab} --unknown)
        set_tests_properties(${lab}_missing_value ${lab}_invalid_repeats ${lab}_unknown_option
            PROPERTIES WILL_FAIL TRUE)
    endforeach()
endif()

install(FILES docs/portable.md DESTINATION . RENAME README.md)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "labs01-03-windows-x64")
set(CPACK_PACKAGE_DIRECTORY "${PROJECT_SOURCE_DIR}/dist")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
include(CPack)
